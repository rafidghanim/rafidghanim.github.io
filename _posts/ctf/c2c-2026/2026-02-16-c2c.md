```js
pragma solidity ^0.8.20;

import {Token} from "./Token.sol";

contract TGE {
    Token public immutable token;

    uint256 public constant TIER_1 = 1;
    uint256 public constant TIER_2 = 2;
    uint256 public constant TIER_3 = 3;

    address public owner;

    mapping(uint256 => uint256) public maxSupply;
    mapping(uint256 => uint256) public mintPrice;

    mapping(uint256 => uint256) public totalSupply;
    mapping(address => mapping(uint256 => uint256)) public balance;

    mapping(uint256 => uint256) public preTGESupply; // snapshot taken ONCE
    mapping(address => mapping(uint256 => uint256)) public preTGEBalance;

    mapping(address => uint256) public userTiers;
    uint256[] public tierIds;

    bool public isTgePeriod;
    bool public tgeActivated;

    modifier onlyOwner() {
        require(msg.sender == owner, "only owner");
        _;
    }

    constructor(
        address _token,
        uint256 maxSupplyTier1,
        uint256 maxSupplyTier2,
        uint256 maxSupplyTier3
    ) {
        require(_token != address(0), "token=0");
        require(maxSupplyTier1 > 0 && maxSupplyTier2 > 0 && maxSupplyTier3 > 0, "bad max");

        token = Token(_token);
        owner = msg.sender;

        maxSupply[TIER_1] = maxSupplyTier1;
        maxSupply[TIER_2] = maxSupplyTier2;
        maxSupply[TIER_3] = maxSupplyTier3;

        mintPrice[TIER_1] = maxSupplyTier1;

        tierIds.push(TIER_1);
        tierIds.push(TIER_2);
        tierIds.push(TIER_3);
    }

    function setTgePeriod(bool _isTge) external onlyOwner {
        if (!_isTge && isTgePeriod && !tgeActivated) {
            tgeActivated = true;
            _snapshotPreTGESupply();
        }

        isTgePeriod = _isTge;
    }

    function buy() external {
        require(isTgePeriod, "TGE closed");
        require(userTiers[msg.sender] == 0, "already registered");

        require(token.transferFrom(msg.sender, address(this), mintPrice[TIER_1]), "payment failed");

        _mint(msg.sender, TIER_1, 1);
        userTiers[msg.sender] = TIER_1;
    }

    function upgrade(uint256 tier) external {
        require(tier <= 3 && tier >= 2);
        require(userTiers[msg.sender]+1 == tier);
        require(tgeActivated && isTgePeriod);

        _burn(msg.sender, tier-1, 1);
        _mint(msg.sender, tier, 1);

        require(preTGEBalance[msg.sender][tier] > preTGESupply[tier], "not eligible");
        userTiers[msg.sender] = tier;
    }

    function _mint(address to, uint256 tier, uint256 quantity) internal {
        _validateTier(tier);
        require(quantity > 0, "qty=0");
        require(totalSupply[tier] + quantity <= maxSupply[tier], "max supply");

        totalSupply[tier] += quantity;
        balance[to][tier] += quantity;

        if (isTgePeriod) {
            preTGEBalance[to][tier] += quantity;
        }
    }

    function _burn(address from, uint256 tier, uint256 quantity) internal {
        _validateTier(tier);
        require(balance[from][tier] >= quantity, "insufficient");

        balance[from][tier] -= quantity;
        totalSupply[tier] -= quantity;
    }

    function _snapshotPreTGESupply() internal {
        for (uint256 i = 0; i < tierIds.length; i++) {
            uint256 id = tierIds[i];
            preTGESupply[id] = totalSupply[id];
        }
    }

    function _validateTier(uint256 tier) internal pure {
        require(tier == TIER_1 || tier == TIER_2 || tier == TIER_3, "invalid tier");
    }
}
```

The smart contract implements a tiered token system with three
levels. Users can upgrade tiers based on token balances and
eligibility checks. The challenge required reaching Tier 3. The
core issue is a logical state flaw in the snapshot mechanism
that governs eligibility. This flaw allows users to bypass
restrictions that would normally prevent them from upgrading
tiers beyond their holdings.
The contract tracks:
- preTGESupply[tier] = global snapshot of total supply, taken
    once when TGE is deactivated.
- preTGEBalance[user][tier] = user’s balance, updated on mint
    while isTgePeriod == true. The upgrade function requires:

![alt text](/assets/img/c2c/image-36.png)

![alt text](/assets/img/c2c/image-37.png)

However, because snapshots are frozen and balances continue to
update, the check can be satisfied.

#### Root Cause

- Snapshot Timing: preTGESupply is set only once. If taken
    before any Tier 2 or 3 tokens exist, the snapshot is 0.
- Ongoing Balance Updates: _mint() updates preTGEBalance for
    a user regardless of snapshot state.\
- Resulting Logic Gap: The contract permits
    [user][tier] > preTGESupply[tier], which should be impossible under normal circumstances.

#### Exploit Strategy

1. Disable TGE → triggers snapshot while Tier 2 and Tier 3 supplies are zero.
2. Re-enable TGE → minting resumes.
3. Buy Tier 1.
4. Upgrade to Tier 2 → preTGEBalance = 1, preTGESupply = 0 → 1
> 0
5. Upgrade to Tier 3 → same logic applies.

By freezing the denominator at zero and increasing the numerator
afterward, the attacker satisfies the eligibility condition and
reaches Tier 3.

| Action       | isTgePeriod | tgeActivated | preTGESupply[2] | preTGEBalance[user][2] |
|-------------|-------------|--------------|-----------------|------------------------|
| Initial     | TRUE        | FALSE        | 0               | 0                      |
| disableTge()| FALSE       | TRUE         | 0               | 0                      |
| enableTge() | TRUE        | TRUE         | 0               | 0                      |
| upgrade(2)  | TRUE        | TRUE         | 0               | 1                      |
| upgrade(3)  | TRUE        | TRUE         | 0               | 1                      |

#### Solution

The challenge involves two smart contracts, Setup and Challenge,
which both decode the same user-supplied bytes input into a
dynamic array of SoulFragment structs and two addresses (invoker
and witness). The goal is to trigger Challenge.transcend() while
satisfying certain constraints.

#### Constraints

1. Setup.bindPact(truth) validates each SoulFragment:
- fragment.essence <= 100 ether for every element
- Records the hash of (user_address, encodedTruth) in chronicles
2. Challenge.transcend(truth) computes:
- totalEssence = sum(fragment.essence)
- Requires totalEssence >= 1000 ether
- Checks that invoker and witness match your wallet address

The core issue is ABI decoding asymmetry:
- Both contracts decode the same bytes truth using abi.decode(truth, (SoulFragment[], address, address)).
- ABI dynamic arrays store offset pointers, and manual
    crafting of these bytes can create a situation where:
    - Setup sees a valid array of fragments with essence <= 100 ether
    - Challenge interprets the same data as a larger array, summing to totalEssence >= 1000 ether

This discrepancy occurs because `abi.decode` trusts the internal
offsets and does not enforce canonical array bounds.

#### Exploit Strategy

1. Register: Call challenge.registerSeeker() to create your account.
2. Craft “Truth” bytes: Manually encode the SoulFragment[] array and addresses to satisfy both contracts:
- Setup sees each fragment ≤100 ether
- Challenge sees total essence ≥1000 ether
- Set invoker and witness to your wallet
3. Seal the Pact: Call setup.bindPact(encodedTruth)
4. Ascend: Call challenge.transcend(encodedTruth) — the ABI interpretation allows transcendence despite Setup’s per-fragment limits.

Since the `/admin` route is explicitly denied by HAProxy before
reaching the backend, direct authentication is clearly not the
intended solution. So, the next step is to inspect the backend
application logic and look for another way to obtain privileged
data.

![alt text](/assets/img/c2c/image.png)

In `utils.py`, the `format_signature()` function exposes the
`current_app` object to user-controlled formatting. If a user can
control the signature template, then `{app.config}` can be used
to dump Flask configuration values.

![alt text](/assets/img/c2c/image-1.png)

After reviewing routes/user.py, we found that the settings
function allows users to define a custom signature. This
signature is later processed by format_signature(), which
applies Python’s str.format() while exposing the current_app
object as app. Since str.format() supports attribute access,
this introduces a format string injection primitive, allowing us
to traverse app attributes and extract sensitive configuration
values such as {app.config}.

After identifying the format string injection primitive, we
replicated the application locally to validate the exploit.
Since user registration is available, we created a regular
account and supplied {app.config} as the signature payload via
the /settings endpoint.

![alt text](/assets/img/c2c/image-2.png)

The application rendered the payload successfully, confirming
that arbitrary attribute traversal was possible. From the
configuration dump, we obtained the JWT_SECRET value. With this
secret key, we were able to forge a valid JWT token and escalate
our privileges.

Although the /admin endpoint is blocked at the HAProxy layer,
further analysis revealed that direct access to this route is
not required to retrieve the flag. After obtaining the
JWT_SECRET via the format string injection, we examined the
token generation logic:

![alt text](/assets/img/c2c/image-3.png)

![alt text](/assets/img/c2c/image-4.png)

Since the server trusts the JWT payload after verifying its
signature, possessing JWT_SECRET allows us to forge arbitrary
tokens. Instead of attempting to access /admin, we impersonated
the user mike.willson (user_id=4).

Ultimately, the attack chain combines configuration disclosure
and token forgery:

* A format string injection exposes `JWT_SECRET`.
* The application fully trusts JWT claims for authorization.
* With the signing key, we can generate arbitrary tokens and
    impersonate any user.
* By forging a token for `mike.willson` (`user_id=4`), we accessed
    the inbox containing the flag

<p style="text-align: center;"><strong>function c(n, r)</strong></p>

4. Caches the result.

- c.decrypt = rc4 assigns the decryptor.

From the runtime dump, the recovered key string is `sulawesi`
and the encryption mode is `AES-CBC`. The first 16 bytes of the
decoded encrypted file are used as the IV, while the remaining
bytes are the ciphertext.

Since AES-CBC keys need to be 8, 16, 32, or 64 bytes and
"sulawesi" is 13 bytes, it can’t be used directly. I noticed
there’s a "SHA-256" thing in the string array that’s not really
used in the code, so I figured the AES key is probably
sha256("sulawesi")[:16] (taking the first 16 bytes because AES
usually wants 16 bytes for AES).
here is my solver

```py
from Crypto.Cipher import AES
import hashlib
from base64 import b64decode

with open('flag.real','r') as f:
    a = f.read()

ct = b64decode(a)[16:]
iv = b64decode(a)[:16]
key = hashlib.sha256(b'sulawesi').digest()[:16]

cipher = AES.new(key,AES.MODE_CBC,iv)
print(cipher.decrypt(ct).decode('latin-1'))
```
